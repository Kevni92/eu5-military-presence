mp_edit_patrol_route = {
	type = owncountry
	show_message = no
	show_in_gui_list = no
	ai_tick = never

	potential = {
		scope:actor = { regular_army_size > 0 }
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = mp_patrol_unit
		name = "mp_edit_patrol_route_select_army"
		none_available_msg_key = "mp_no_available_armies"
		column = { data = name }
		column = { data = unit_location }
		visible = {
			is_army = yes
			owner = scope:actor
			NOT = { has_variable = mp_patrol_active }
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = mp_patrol_editor_focus_province
		name = "mp_edit_patrol_route_select_provinces"
		none_available_msg_key = "mp_no_available_patrol_provinces"
		tooltip_msg_key = "mp_patrol_editor_map_tt"
		move_to_next_section_on_click = no
		top_widget = mp_patrol_route_editor_top
		bottom_widget = mp_patrol_route_editor_bottom
		column = { data = mp_patrol_route_toggle }
		column = { data = name }
		column = { data = population }
		visible = {
			any_location_in_province = {
				owner = scope:actor
				count = all
			}
			scope:mp_patrol_unit = {
				owner = scope:actor
				is_army = yes
				NOT = { has_variable = mp_patrol_active }
			}
		}
		selected = {
			scope:mp_patrol_unit = {
				OR = {
					AND = {
						has_variable = mp_patrol_edit_open
						is_target_in_variable_list = {
							name = mp_patrol_edit_provinces
							target = root
						}
					}
					AND = {
						NOT = { has_variable = mp_patrol_edit_open }
						is_target_in_variable_list = {
							name = mp_patrol_provinces
							target = root
						}
					}
				}
			}
		}

		# map_color executes on Location scope. Selection state is stored on the
		# containing Province so the renderer does not need to dereference a unit.
		map_color = {
			if = {
				limit = {
					province = {
						has_variable = mp_patrol_editor_selected_map
					}
				}
				value = rgb { 55 155 95 }
			}
			else = {
				value = rgb { 90 90 90 }
			}
		}
		secondary_map_color = {
			if = {
				limit = {
					province = {
						has_variable = mp_patrol_editor_selected_map
					}
				}
				value = rgb { 230 190 65 }
			}
		}
		only_color_selectable = yes
	}

	effect = {
	}

	ai_will_do = { value = -1 }
}

mp_start_military_patrol = {
	type = owncountry
	show_message = no
	show_in_gui_list = no
	ai_tick = never

	potential = {
		scope:actor = { regular_army_size > 0 }
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = mp_patrol_unit
		name = "mp_start_military_patrol_select_army"
		none_available_msg_key = "mp_no_configured_patrol_armies"
		column = { data = name }
		column = { data = unit_location }
		visible = {
			is_army = yes
			owner = scope:actor
			has_variable = mp_patrol_route_configured
			has_variable_list = mp_patrol_provinces
			NOT = { has_variable = mp_patrol_active }
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:mp_patrol_unit
				scope:mp_patrol_unit = {
					owner = scope:actor
					is_army = yes
					has_variable = mp_patrol_route_configured
					has_variable_list = mp_patrol_provinces
					NOT = { has_variable = mp_patrol_active }
				}
			}
			scope:mp_patrol_unit = {
				set_variable = { name = mp_patrol_active value = 1 }
			}
			scope:actor = {
				mp_refresh_military_presence_assignments_effect = yes
			}
		}
	}

	ai_will_do = { value = -1 }
}

mp_stop_military_patrol = {
	type = owncountry
	show_message = no
	show_in_gui_list = no
	ai_tick = never

	potential = {
		scope:actor = { regular_army_size > 0 }
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = mp_patrol_unit
		name = "mp_stop_military_patrol_select_army"
		none_available_msg_key = "mp_no_active_patrol_armies"
		column = { data = name }
		column = { data = unit_location }
		visible = {
			is_army = yes
			owner = scope:actor
			has_variable = mp_patrol_active
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:mp_patrol_unit
				scope:mp_patrol_unit = {
					owner = scope:actor
					is_army = yes
					has_variable = mp_patrol_active
				}
			}
			scope:mp_patrol_unit = {
				remove_variable = mp_patrol_active
			}
			scope:actor = {
				mp_refresh_military_presence_assignments_effect = yes
			}
		}
	}

	ai_will_do = { value = -1 }
}
