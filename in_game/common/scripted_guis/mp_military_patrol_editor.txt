# Scripted-GUI bridge for the Military Patrol multi-province editor.
# Root scope is the player country. Unit/province are passed as saved scopes.

mp_patrol_editor_begin = {
	scope = country
	saved_scopes = { mp_patrol_unit }
	is_valid = {
		scope:mp_patrol_unit ?= { is_army = yes owner = root NOT = { has_variable = mp_patrol_active } }
	}
	effect = {
		if = {
			limit = { scope:mp_patrol_unit ?= { is_army = yes owner = root NOT = { has_variable = mp_patrol_active } } }
			scope:mp_patrol_unit = {
				remove_variable = mp_patrol_editor_ignore_next_focus
				mp_begin_patrol_route_editor_effect = yes
			}
		}
	}
}

# Table actions also call vanilla SelectInteractionTargetGlue.OnLeftClick after
# this scripted action so that the selection map gets the same native focus/
# redraw signal as an ordinary target-row click. If that callback will move the
# current focus, suppress exactly that next focus event in the additive map-click
# watcher. mp_patrol_editor_last_map_focus is our current focus mirror.
mp_patrol_editor_toggle_province = {
	scope = country
	saved_scopes = { mp_patrol_unit mp_patrol_province }
	is_valid = {
		scope:mp_patrol_unit ?= { is_army = yes owner = root has_variable = mp_patrol_edit_open NOT = { has_variable = mp_patrol_active } }
		scope:mp_patrol_province ?= { any_location_in_province = { owner = root count = all } }
	}
	effect = {
		if = {
			limit = {
				scope:mp_patrol_unit ?= { is_army = yes owner = root has_variable = mp_patrol_edit_open NOT = { has_variable = mp_patrol_active } }
				scope:mp_patrol_province ?= { any_location_in_province = { owner = root count = all } }
			}
			scope:mp_patrol_unit = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable = mp_patrol_editor_last_map_focus }
							var:mp_patrol_editor_last_map_focus ?= { NOT = { this = scope:mp_patrol_province } }
						}
					}
					set_variable = { name = mp_patrol_editor_ignore_next_focus value = 1 }
				}
				mp_toggle_patrol_route_editor_province_effect = yes
			}
		}
	}
}

mp_patrol_editor_map_focus_changed = {
	scope = country
	saved_scopes = { mp_patrol_unit mp_patrol_province }
	is_valid = {
		scope:mp_patrol_unit ?= {
			is_army = yes owner = root has_variable = mp_patrol_edit_open NOT = { has_variable = mp_patrol_active }
			OR = {
				NOT = { has_variable = mp_patrol_editor_last_map_focus }
				var:mp_patrol_editor_last_map_focus ?= { NOT = { this = scope:mp_patrol_province } }
			}
		}
		scope:mp_patrol_province ?= { any_location_in_province = { owner = root count = all } }
	}
	effect = {
		if = {
			limit = {
				scope:mp_patrol_unit ?= { has_variable = mp_patrol_edit_open }
				scope:mp_patrol_province ?= { any_location_in_province = { owner = root count = all } }
			}
			scope:mp_patrol_unit = {
				if = {
					limit = { has_variable = mp_patrol_editor_ignore_next_focus }
					remove_variable = mp_patrol_editor_ignore_next_focus
					set_variable = { name = mp_patrol_editor_last_map_focus value = scope:mp_patrol_province }
				}
				else = {
					mp_note_patrol_route_editor_map_focus_effect = yes
				}
			}
		}
	}
}

mp_patrol_editor_province_selected = {
	scope = country
	saved_scopes = { mp_patrol_unit mp_patrol_province }
	is_shown = {
		scope:mp_patrol_unit ?= {
			OR = {
				AND = { has_variable = mp_patrol_edit_open is_target_in_variable_list = { name = mp_patrol_edit_provinces target = scope:mp_patrol_province } }
				AND = { NOT = { has_variable = mp_patrol_edit_open } is_target_in_variable_list = { name = mp_patrol_provinces target = scope:mp_patrol_province } }
			}
		}
	}
	effect = { }
}
