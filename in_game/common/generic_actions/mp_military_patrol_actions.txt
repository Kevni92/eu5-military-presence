mp_add_patrol_province = {
	type = owncountry
	show_message = no
	ai_tick = never

	potential = {
		scope:actor = {
			regular_army_size > 0
		}
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = mp_patrol_unit
		name = "mp_add_patrol_province_select_army"
		none_available_msg_key = "mp_no_available_armies"
		column = { data = name }
		column = { data = unit_location }
		visible = { is_army = yes }
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = mp_patrol_province
		name = "mp_add_patrol_province_select_province"
		none_available_msg_key = "mp_no_available_patrol_provinces"
		column = { data = name }
		column = { data = population }
		visible = {
			any_location_in_province = {
				owner = scope:actor
				count = all
			}
			scope:mp_patrol_unit = {
				NOT = {
					is_target_in_variable_list = {
						name = mp_patrol_provinces
						target = root
					}
				}
			}
		}
	}

	effect = {
		scope:mp_patrol_unit = {
			mp_add_patrol_province_effect = yes
		}
	}

	ai_will_do = { value = -1 }
}

# Scripted army assignment replacing the former Unit Ability.
# This remains an ordinary script state; it does not register a new hardcoded
# military objective type such as Carpet Siege.
mp_start_military_patrol = {
	type = owncountry
	show_message = no
	ai_tick = never

	potential = {
		scope:actor = { regular_army_size > 0 }
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = unit
		source = actor
		target_flag = mp_patrol_unit
		name = "mp_start_military_patrol_select_army"
		none_available_msg_key = "mp_no_configured_patrol_armies"
		column = { data = name }
		column = { data = unit_location }
		visible = {
			is_army = yes
			has_variable_list = mp_patrol_provinces
			NOT = { has_variable = mp_patrol_active }
		}
	}

	effect = {
		scope:mp_patrol_unit = {
			set_variable = { name = mp_patrol_active value = 1 }
		}
	}

	ai_will_do = { value = -1 }
}
