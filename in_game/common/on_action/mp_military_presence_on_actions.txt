monthly_country_pulse = {
	on_actions = {
		mp_military_presence_monthly_pulse
	}
}

mp_military_presence_monthly_pulse = {
	trigger = {
		OR = {
			any_army = {
				is_exiled = no
			}
			any_province = {
				has_variable = mp_military_presence
			}
		}
	}

	effect = {
		mp_monthly_country_presence_effect = yes

		# Rebuild the display cache after Presence processing so a patrol that
		# advances this month is striped on its new current target immediately.
		mp_refresh_military_presence_assignments_effect = yes
	}
}
